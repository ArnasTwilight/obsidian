В `Controller_Batch->action_generate_bills()` есть генерация счетов `Model_Supplierdoc->generate_bill()`, если номенклатура равна 10, то происходит создание счёта и сервиса.
Нужно встроиться в метод и выполнить проверку по условиям:
1. Проверить lcode по списку в ТЗ
2. Проверить количество транзакций, если они закончились, то нужно:
	 - Закрыть договор на шлюз (10)
	 - Открыть договор на шлюз 0,5 (48)
	 - Перевод наполнения договора на новую номенклатуру. (Информация по платёжному шлюзу хранится в onlinepaymentv2hotelsystem и её поиск ведётся по Lcode без привязки к номенклатуре)

Далее необходимо подправить текущий процесс формирования счетов для переведённых клиентов. 
В разделе "Комиссия" -> "Выставление счетов за Шлюз оплат 0,5" -> Загружаем файл с данными для создания счёта. После сталкиваемся с проблемами:
- Мы не можем выставить счёт по шлюзу 0.5, если у нас номенклатура не 46 и 47 
- Мы не можем выставить счёт по шлюзу 0.5, если у нас нет баланса по этому
Для реализации необходимо обратиться к `Controller_Commission->action_form_payment_gateway_05_bills()` далее обращаемся к `Services_Commissionbyproducttypesservice->processCommissionFile()->generateBillsForCommission()` там есть разделение на шлюз оплат и новые продукты перед выставлением счетов, там и необходимо внести правки на соответствие условиям:
1. Совпадает по lcode из ТЗ
2. Разрешаем создание счетов для старых номенклатур
3. Игнорируем отсутствие баланса
Далее создаём счёт в статусе "не оплачен".

Хранение lcode из ТЗ предлагаю в .json в директории: `app/application/files/array.json`


```php
<?php  
  
class Services_Commissionbyproducttypesservice  
{  
   private $serviceType;  
   const PRODUCT_TYPE_TO_TYPE = [  
      'Bnovo PMS / Комплекс' => Model_Supplierdoctype::SDTYPE_BNOVO_PMS_COMPLEX,  
      'ЧМ + Модуль / Онлайн-продажи' => Model_Supplierdoctype::SDTYPE_CHM_MODULE_ONLINE_SALES,  
   ];  
   const SERVICE_TYPE_NEW_PRODUCTS = 1;  
   const SERVICE_TYPE_PAYMENT_GATEWAY_05 = 2;  
  
  
   public function __construct($serviceType = self::SERVICE_TYPE_NEW_PRODUCTS)  
   {  
      $this->serviceType = $serviceType;  
   }  
  
   public function processCommissionFile($filePath)  
   {  
      $data = [];  
      $errorsData = [  
         'rows' => [],  
         'fields' => []  
      ];  
      $fileData = $this->parse_file($filePath);  
      foreach ($fileData as $rowNumber => $row) {  
         try {  
            $commissionRow = [  
               'row_num' => $rowNumber,  
               'hotel_id' => $row[0],  
               'date_start' => $row[1],  
               'date_end' => $row[2],  
            ];  
  
            if ($this->isForNewProducts()) {  
               $commissionRow['product_type'] = $this->matchProductTypeToDocType($row[3]);  
               $commissionRow['sum'] = $row[5];  
            } elseif ($this->isForPaymentGateway05()) {  
               $commissionRow['sum'] = $row[3];  
               $commissionRow['pms_id'] = $row[4];  
            }  
     
            $this->validateSingleRow($commissionRow, $errorsData);  
  
            $data[] = $commissionRow;  
  
            if (count($errorsData['rows']) > 0 || count($errorsData['fields']) > 0) {  
               $rows = implode(', ', $errorsData['rows']);  
               $fields = implode(', ', $errorsData['fields']);  
               $errors[] = 'В загруженном документе не хватает исходных данных. В строках ' . $rows . ' не заполнены данные о ' . $fields . ' или заполнены некорректно';  
           return ['errors' => $errors];  
         }  
         } catch (Exception $e) {  
            Kohana::$log->add('ERROR', 'Некорректный файл!');  
            $errors[] = 'Некорректный файл!';  
  
         return ['errors' => $errors];  
       }  
      }  
  
      if ($this->isForNewProducts()) {  
         $this->generateBillsForCommission($data);  
      } elseif ($this->isForPaymentGateway05()) {  
         $this->generateBillsForPaymentGateway05($data);  
      }  
  
     return $this->getProcessSuccessResponse();  
   }  
  
   private function generateBillsForCommission(array $data)  
   {  
      foreach ($data as $commissionRow) {  
         if ($commissionRow['sum'] == 0) {  
            Kohana::$log->add('INFO', 'Для отеля: ' . $commissionRow['hotel_id'] . ' с номенклатурой: '  
               . $commissionRow['product_type'] . ' комиссия равна 0, пропускаем');  
            continue;         }  
  
         $supplierDoc = $this->getSupplierDoc($commissionRow['hotel_id'], $commissionRow['product_type']);  
  
         if (!$supplierDoc->loaded()) {  
            Kohana::$log->add('ERROR', 'Не найден договор для отеля: ' . $commissionRow['hotel_id']  
               . ' с номенклатурой: ' . $commissionRow['product_type']  
               . ', статус: ' . Model_Supplierdocstatus::STATUS_CONTRACT_CONCLUDED);  
            continue;         }  
  
         $balance = $this->getBalance($supplierDoc->pk());  
         if (!$balance->loaded()) {  
            Kohana::$log->add('ERROR', 'Баланс для договора ID ' . $supplierDoc->pk() . ' не найден');  
            continue;         }  
  
         $resultSum = $balance->balance - $commissionRow['sum'];  
         $this->definitionBalanceAndCreateBill($resultSum, $commissionRow, $supplierDoc, $balance);  
      }  
   }  
  
   /**  
    * Генерирует счета для "Шлюз оплат 0,5" (48)    * @param array $data  
    * @return void  
    */   private function generateBillsForPaymentGateway05(array $data)  
   {  
      foreach ($data as $commissionRow) {  
         if ($commissionRow['sum'] == 0) {  
            Kohana::$log->add('INFO', 'Для отеля: ' . $commissionRow['hotel_id'] . ' с номенклатурой: '  
               . Model_Supplierdoctype::SDTYPE_PAYMENT_GATEWAY_05 . ' комиссия равна 0, пропускаем');  
            continue;         }  
  
         $supplierDoc = $this->getSupplierDoc($commissionRow['hotel_id'], Model_Supplierdoctype::SDTYPE_PAYMENT_GATEWAY_05);  
         if (!$supplierDoc->loaded()) {  
            Kohana::$log->add('ERROR', 'Не найден договор для отеля: ' . $commissionRow['hotel_id']  
               . ' с номенклатурой ' . Model_Supplierdoctype::SDTYPE_PAYMENT_GATEWAY_05);  
            continue;         }  
  
         if (!$this->checkLcodeInGatewayTransferred($commissionRow['pms_id'])) {  
            $this->generateBillsForStandardScenario($commissionRow, $supplierDoc);  
         } else {  
            $this->generateBillsForTransferredClient($commissionRow, $supplierDoc);  
         }  
      }  
   }  
  
   /**  
    * Создаёт счёт в статусе не оплачен и сервис счёта для переведённых    * клиентов с "Шлюз оплат" (10) на "Шлюз оплат 0,5" (48)    * @param $commissionRow  
    * @param $supplierDoc  
    * @return void  
    */   private function generateBillsForTransferredClient($commissionRow, $supplierDoc)  
   {  
      $supplierDocCommissionDto = new Dto_Supplierdoccommissiondto($commissionRow['sum'], $commissionRow['date_start'], $commissionRow['date_end']);  
  
      $billDataUnpaid = $this->prepareBillData($supplierDoc->pk(), $supplierDocCommissionDto, Model_Bill::STATUS_ID_UNPAID);  
      $billUnpaid = $this->createBill($billDataUnpaid);  
      $this->createServiceForBill($supplierDoc, $billUnpaid);  
   }  
  
   /**  
    * Создаёт счета по стандартному сценарию для "Шлюз оплат 0,5" (48)    * @param array $commissionRow  
    * @param $supplierDoc  
    * @return bool  
    */   private function generateBillsForStandardScenario(array $commissionRow, $supplierDoc)  
   {  
      $mainSupplierDoc = ORM::factory('supplierdoc')  
         ->where('lcode', '=', $commissionRow['pms_id'])  
         ->where('type_id', 'IN', [  
            Model_Supplierdoctype::SDTYPE_BNOVO_PMS_COMPLEX,  
            Model_Supplierdoctype::SDTYPE_CHM_MODULE_ONLINE_SALES  
         ])  
         ->find();  
  
      if (!$mainSupplierDoc->loaded()) {  
         Kohana::$log->add('ERROR', 'Не найден договор на основной продукт для отеля: '  
            . $commissionRow['hotel_id'] . ' с номенклатурой ' . Model_Supplierdoctype::SDTYPE_PAYMENT_GATEWAY_05);  
         return false;      }  
  
      if (!$this->checkSupplierDocGatewayRelation($supplierDoc->pk())) {  
         $dataSupplierDoc = [  
            'supplierdoc_id' => $mainSupplierDoc->pk(),  
            'pms_id' => $mainSupplierDoc->lcode,  
         ];  
         $this->createSupplierDocGatewayRelation($dataSupplierDoc);  
      }  
  
      $balance = ORM::factory('supplierdocbalance')  
         ->where('supplierdoc_id', '=', $mainSupplierDoc->pk())  
         ->find();  
      if (!$balance->loaded()) {  
         Kohana::$log->add('ERROR', 'Баланс для договора ID ' . $supplierDoc->pk() . ' не найден');  
         return false;      }  
  
      $resultSum = $balance->balance - $commissionRow['sum'];  
      $this->definitionBalanceAndCreateBill($resultSum, $commissionRow, $supplierDoc, $balance);  
  
      return true;   }  
  
   private function createBill(array $billData)  
   {  
      /** @var Model_Bill $bill */  
      return ORM::factory('bill')  
         ->values($billData)  
         ->save();  
   }  
  
   /**  
    * Подготовка данных для создания счёта в зависимости от статуса    * @param $supplierDocId  
    * @param Dto_Supplierdoccommissiondto $supplierDocCommissionDto  
    * @param $statusId  
    * @return array  
    */   private function prepareBillData($supplierDocId, Dto_Supplierdoccommissiondto $supplierDocCommissionDto, $statusId)  
   {  
      $billData = [  
         'supplierdoc_id' => $supplierDocId,  
         'amount' => $supplierDocCommissionDto->getSum(),  
         'date_start' => $supplierDocCommissionDto->getDateStart(),  
         'date_end' => $supplierDocCommissionDto->getDateEnd(),  
         'status_id' => $statusId,  
         'user_id' => 1,  
         'univercity' => 'N',  
         'payment_type' =>  Model_Bill::PAYMENT_TYPE_POSTPAID,  
         'show_act' => 'Y',  
      ];  
  
      if ($statusId == Model_Bill::STATUS_ID_UNPAID) {  
         $billData['date_product_expire'] = date('Y-m-d', strtotime('+14 days'));  
      }  
  
      return $billData;  
   }  
  
   private function createServiceForBill(Model_Supplierdoc $supplierDoc, Model_Bill $bill)  
   {  
      $service = ORM::factory('billservice');  
  
      $service->supplierbill_id = $bill->pk();  
      $service->service_id = $supplierDoc->get_service_by_type($supplierDoc->type->pk());  
      $service->quantity = 1;  
      $service->price = $bill->amount;  
  
      $service->save();  
   }  
  
   /**  
    * @param array $dataSupplierDoc  
    * @return ORM  
    */   private function createSupplierDocGatewayRelation(array $dataSupplierDoc)  
   {  
      return ORM::factory('supplierdocgatewayrelation')  
         ->values($dataSupplierDoc)  
         ->save();  
   }  
  
   private function parse_file($file) {  
      $result = [];  
  
      $index = 0;  
      $fp = fopen($file, 'r');  
      if ($fp) {  
         while ($row = fgetcsv($fp, '', ';')) {  
            if ($index == 0) {  
               $index++;  
               continue;            }  
            for ($i = 0; $i < count($row); $i++) {  
               $row[$i] = mb_convert_encoding($row[$i], 'UTF-8', 'UTF-8');  
            }  
  
            $result[$index] = $row;  
            $index++;  
         }  
  
         fclose($fp);  
      }  
  
      return $result;  
   }  
  
   private function validateSingleRow($commissionRow, &$errorsData)  
   {  
      foreach ($commissionRow as $field => $value) {  
         if (empty($value)) {  
            if (!in_array($commissionRow['row_num'], $errorsData['rows'])) {  
               $errorsData['rows'][] = $commissionRow['row_num'];  
            }  
            if (!in_array($field, $errorsData['fields'])) {  
               $errorsData['fields'][] = $field;  
            }  
            if ($this->isForNewProducts()) {  
               Kohana::$log->add('ERROR', 'Ошибка при создании счета на постоплату, строка: ' . $commissionRow['row_num']);  
            } elseif ($this->isForPaymentGateway05()) {  
               Kohana::$log->add('ERROR', 'Ошибка при создании счёта, строка: ' . $commissionRow['row_num']);  
            }  
         }  
      }  
   }  
  
   private function matchProductTypeToDocType($productType)  
   {  
      return array_key_exists($productType, self::PRODUCT_TYPE_TO_TYPE) ? self::PRODUCT_TYPE_TO_TYPE[$productType] : null;  
   }  
  
   private function makeNegativeNumber($number)  
   {  
      return -1 * $number;  
   }  
  
   private function isForNewProducts()  
   {  
      return $this->serviceType == self::SERVICE_TYPE_NEW_PRODUCTS;  
   }  
  
   private function isForPaymentGateway05()  
   {  
      return $this->serviceType == self::SERVICE_TYPE_PAYMENT_GATEWAY_05;  
   }  
  
   /**  
    * Выполняет проверку на существование записи в Model_Supplierdocgatewayrelation    * @param $supplierDocId  
    * @return bool  
    */   private function checkSupplierDocGatewayRelation($supplierDocId)  
   {  
      $supplierDocGatewayRelation = ORM::factory('supplierdocgatewayrelation')  
         ->where('supplierdoc_id', '=', $supplierDocId)  
         ->find();  
  
      if ($supplierDocGatewayRelation->loaded()) {  
         return true;  
      }  
  
      return false;  
   }  
  
   /**  
    * Проверяет lcode на присутствие в массиве с клиентами, которые были переведены на "Шлюз оплат 0,5" (48)    * @param $checkLcode  
    * @return bool  
    */   private function checkLcodeInGatewayTransferred($checkLcode)  
   {  
      $filePath = Kohana::config('site')['file_path_lcode_list_transfer_gateway'];  
      $lcodeList = (new Services_Getjsonfile)->getJsonFromFileToArray($filePath);  
  
      if (in_array($checkLcode, $lcodeList)) {  
         return true;  
      }  
  
      return false;  
   }  
  
   /**  
    * Создание счёта на основе данных из баланса    * @param $resultSum  
    * @param $commissionRow  
    * @param $supplierDoc  
    * @param $balance  
    * @return void  
    */   private function definitionBalanceAndCreateBill($resultSum, $commissionRow, $supplierDoc, $balance)  
   {  
      $SupplierDocBalanceService = new Services_Supplierdocbalanceservice();  
      $sum = $commissionRow['sum'];  
      $dateStart = $commissionRow['date_start'];  
      $dateEnd = $commissionRow['date_end'];  
  
      if ($resultSum > 0 || $resultSum == 0) {  
         $supplierDocCommissionDto = new Dto_Supplierdoccommissiondto($sum, $dateStart, $dateEnd);  
  
         $billDataPaid = $this->prepareBillData($supplierDoc->pk(), $supplierDocCommissionDto, Model_Bill::STATUS_ID_DEPOSIT_PAID);  
  
         $billPaid = $this->createBill($billDataPaid);  
         $this->createServiceForBill($supplierDoc, $billPaid);  
         $SupplierDocBalanceService->changeBalance($billPaid, $this->makeNegativeNumber($supplierDocCommissionDto->getSum()));  
  
      } else if ($resultSum < 0 && $balance->balance > 0) {  
         $supplierDocCommissionPaidBillDto = new Dto_Supplierdoccommissiondto($balance->balance, $dateStart, $dateEnd);  
         $supplierDocCommissionUnpaidBillDto = new Dto_Supplierdoccommissiondto(abs($resultSum), $dateStart, $dateEnd);  
  
         $billDataPaid = $this->prepareBillData($supplierDoc->pk(), $supplierDocCommissionPaidBillDto, Model_Bill::STATUS_ID_DEPOSIT_PAID);  
         $billDataUnpaid = $this->prepareBillData($supplierDoc->pk(), $supplierDocCommissionUnpaidBillDto, Model_Bill::STATUS_ID_UNPAID);  
  
         $billPaid = $this->createBill($billDataPaid);  
         $billUnpaid = $this->createBill($billDataUnpaid);  
  
         $this->createServiceForBill($supplierDoc, $billPaid);  
         $this->createServiceForBill($supplierDoc, $billUnpaid);  
  
         $SupplierDocBalanceService->changeBalance($billPaid, $this->makeNegativeNumber($supplierDocCommissionPaidBillDto->getSum()));  
         $SupplierDocBalanceService->changeBalance($billUnpaid, $this->makeNegativeNumber($supplierDocCommissionUnpaidBillDto->getSum()));  
  
      } else if ($resultSum < 0 && $balance->balance <= 0) {  
         $supplierDocCommissionDto = new Dto_Supplierdoccommissiondto($sum, $dateStart, $dateEnd);  
  
         $billDataUnpaid = $this->prepareBillData($supplierDoc->pk(), $supplierDocCommissionDto, Model_Bill::STATUS_ID_UNPAID);  
  
         $billUnpaid = $this->createBill($billDataUnpaid);  
         $this->createServiceForBill($supplierDoc, $billUnpaid);  
         $SupplierDocBalanceService->changeBalance($billUnpaid, $this->makeNegativeNumber($supplierDocCommissionDto->getSum()));  
      }  
   }  
  
   private function getProcessSuccessResponse() {  
      if ($this->isForPaymentGateway05()) {  
         return ['success' => 'Счета на Шлюз оплат 0,5 были успешно созданы!'];  
      }   
        
      return  ['success' => 'Счета на постоплату были успешно созданы!'];  
   }  
  
   /**  
    * Возвращает модель номенклатуры в статусе "Заключён" (4)    * @param $commissionRow  
    * @param $typeId  
    * @return Model_Supplierdoc  
    */   private function getSupplierDoc($lcode, $typeId)  
   {  
      return ORM::factory('supplierdoc')  
         ->where('lcode', '=', $lcode)  
         ->where('type_id', '=', $typeId)  
         ->where('status_id', '=', Model_Supplierdocstatus::STATUS_CONTRACT_CONCLUDED)  
         ->find();  
   }  
  
   /**  
    * @param int $supplierDocId  
    * @return Model_Supplierdocbalance  
    */   private function getBalance($supplierDocId)  
   {  
      return ORM::factory('supplierdocbalance')  
         ->where('supplierdoc_id', '=', $supplierDocId)  
         ->find();  
   }  
}
```
